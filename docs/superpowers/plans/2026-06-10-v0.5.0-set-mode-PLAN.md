# cache-necromancer v0.5.0 set 소생 모드 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 기본 동작을 "만료 임박 알림만"으로 바꾸고, 자동 wake 는 사용자가 `/cn:set N` 으로 명시 충전한 예산만큼만 일어나게 한다 (spec: `docs/superpowers/specs/active/2026-06-10-cache-necromancer-v0.5.0-set-mode-design.md`).

**Architecture:** `mode` enum (notify/auto/hybrid) 을 알림(`notify.enabled`) / wake(`wake.arm` = manual|always) 2축으로 분해한다. wake 예산은 per-session marker 에 3개 필드(`set_budget_remaining/total`, `set_charged_at_ns`)로 저장하고, `refresh.py` 가 sleep 후 예산을 보고 wake 여부를 결정한다. `/cn:set` 은 기존 `/cn:status` 와 같은 UserPromptExpansion 패턴(LLM turn 0회)으로 처리한다. 기존 config 는 로드 시 매핑으로 동작 보존.

**Tech Stack:** Python 3.11+ stdlib only (기존과 동일), pytest + freezegun (`uv run --extra dev pytest`).

**전환 전략 (suite green 유지):** Task 2 에서 `Config` 에 신 필드를 넣되 구 코드가 쓰는 `config.mode` / `config.refresh.hybrid_wait_seconds` 를 **임시 호환 property** 로 유지한다. Task 3~8 에서 소비처를 신 필드로 옮기고, Task 9 에서 임시 property 를 제거한다. 각 Task 종료 시 전체 suite 가 green 이어야 한다.

VALID_ARMS: tuple[str, ...] = ("manual", "always")
_LEGACY_MODES: tuple[str, ...] = ("notify", "auto", "hybrid")

# legacy 키 (v0.4.x 이하) — 로드시 매핑, /cn:status 경고용
_LEGACY_GENERAL = {"mode"}
_LEGACY_NOTIFY = {"system_notification"}
_LEGACY_REFRESH = {"hybrid_wait_seconds"}


@dataclass(frozen=True)
class WakeConfig:
    arm: str = "manual"
    grace_seconds: int = 60


@dataclass(frozen=True)
class NotifyConfig:
    enabled: bool = True


@dataclass(frozen=True)
class RefreshConfig:  # 임시 호환 (Task 9 제거): config.refresh.hybrid_wait_seconds
    hybrid_wait_seconds: int = 60


@dataclass(frozen=True)
class Config:
    refresh_interval_minutes: int = 50
    cache_ttl_minutes: int = 60
    max_refresh_count: int = 10
    language: str = "en"
    wake: WakeConfig = field(default_factory=WakeConfig)
    notify: NotifyConfig = field(default_factory=NotifyConfig)

    # ── 임시 호환 property (Task 9 에서 제거) ──
    @property
    def mode(self) -> str:
        if self.wake.arm == "always":
            return "hybrid" if self.notify.enabled else "auto"
        return "notify"

    @property
    def refresh(self) -> RefreshConfig:
        return RefreshConfig(hybrid_wait_seconds=self.wake.grace_seconds)


def detect_legacy_keys(data: dict) -> list[str]:
    """v0.4.x 이하 legacy 키 목록 (매핑은 되지만 /cn:status 에서 경고)."""
    found: list[str] = []
    for key in data.get("general", {}):
        if key in _LEGACY_GENERAL:
            found.append(f"general.{key}")
    for key in data.get("notify", {}):
        if key in _LEGACY_NOTIFY:
            found.append(f"notify.{key}")
    for key in data.get("refresh", {}):
        if key in _LEGACY_REFRESH:
            found.append(f"refresh.{key}")
    return found


def _resolve_axes(data: dict) -> tuple[WakeConfig, NotifyConfig]:
    """신 키 우선, 없으면 legacy 매핑 (spec §3.4), 그것도 없으면 기본값."""
    general = data.get("general", {})
    wake_data = data.get("wake", {})
    notify_data = data.get("notify", {})

    # legacy 해석
    legacy_arm: str | None = None
    legacy_enabled: bool | None = None
    mode = general.get("mode")
    if mode is not None:
        if mode in _LEGACY_MODES:
            legacy_arm = "always" if mode in ("hybrid", "auto") else "manual"
            # 구 auto 는 system_notification 과 무관하게 알림 없이 즉시 wake
            legacy_enabled = mode != "auto"
        else:
            print(
                f"[cn:warn] invalid legacy mode: {mode!r} — 무시 (기본값 사용)",
                file=sys.stderr,
            )
    if legacy_enabled is not False and "system_notification" in notify_data:
        legacy_enabled = bool(notify_data["system_notification"])
    legacy_grace = data.get("refresh", {}).get("hybrid_wait_seconds")

    # 신 키 우선
    arm = wake_data.get("arm", legacy_arm if legacy_arm is not None else "manual")
    if arm not in VALID_ARMS:
        print(
            f"[cn:warn] invalid wake.arm: {arm!r} — fallback to 'manual'",
            file=sys.stderr,
        )
        arm = "manual"
    grace = wake_data.get(
        "grace_seconds", legacy_grace if legacy_grace is not None else 60
    )
    enabled = notify_data.get(
        "enabled", legacy_enabled if legacy_enabled is not None else True
    )
    return WakeConfig(arm=arm, grace_seconds=int(grace)), NotifyConfig(
        enabled=bool(enabled)
    )
```

def ensure_config_file(path: Path) -> None:
    """파일이 없으면 기본 템플릿 작성. 있으면 그대로 둔다 (사용자 편집 보존).

        return
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(_DEFAULT_TEMPLATE, encoding="utf-8")
    try:
        os.chmod(path, 0o600)
    except OSError:
        pass

### Task 3: refresh.py — arm/예산 분기 + ping (consumed/total)

**Files:**
- Modify: `scripts/refresh.py`
- Test: `tests/scripts/test_refresh.py`

동작 (spec §7):
- 진입부 `max_refresh_count` skip 은 **always 일 때만** (manual 은 예산이 상한).
- manual + 예산 0 이어도 **sleep 은 한다** — sleep 중 `/cn:set` 충전을 소비해야 하므로 (codex F1 핵심).

def _do_wake(marker: Marker, sid_hash: str, config: Config) -> int:
    """wake_count/예산 갱신 + save 후 stderr ping + exit 2.

    manual(예산 소비) 이면 ping 은 (consumed/total), always 는 (count/max).
    save 실패 시 wake 자체는 진행 (cache 갱신 우선, count 누락 허용).
    """
    marker.wake_count += 1
    marker.last_wake_at = int(time.time())
    if marker.set_budget_remaining > 0:
        consumed = marker.set_budget_total - marker.set_budget_remaining + 1
        marker.set_budget_remaining -= 1
        nm = f"{consumed}/{marker.set_budget_total}"
    else:
        nm = f"{marker.wake_count}/{config.max_refresh_count}"
    _save_marker(marker, "wake")
    log_info(f"[refresh] wake sid={sid_hash} count={marker.wake_count} nm={nm}")
    print(_build_ping(nm), file=sys.stderr)
    return 2

#!/usr/bin/env python3
"""/cn:set backend (spec §4) — 현재 세션 wake 예산 충전/취소/조회.

UserPromptExpansion(on_status_command.py) 또는 commands/cn:set.md fallback 이
argv 로 인자를 넘긴다. 출력은 stdout plain text (LLM turn 0회).
"""
import os
import sys
import time
from datetime import datetime, timedelta
from pathlib import Path

_HERE = Path(__file__).resolve().parent
_PROJECT_ROOT = _HERE.parent
if str(_PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(_PROJECT_ROOT))

from lib.config import Config, ensure_config_file, load_config  # noqa: E402
from lib.i18n import normalize_language, set_label  # noqa: E402
from lib.marker import Marker  # noqa: E402
from lib.session_id import sanitize  # noqa: E402


def _resolve_root() -> Path:
    root = os.environ.get("CN_ROOT")
    return Path(root) if root else Path.home() / ".cache-necromancer"


def _survive_time(config: Config, remaining: int) -> str:
    at = datetime.now() + timedelta(
        minutes=remaining * config.refresh_interval_minutes + config.cache_ttl_minutes
    )
    return at.strftime("%H:%M")


def main(argv: list[str]) -> int:
    sid = os.environ.get("CLAUDE_CODE_SESSION_ID", "")
    if not sid:
        print("session_id 없음 — chat 세션 안에서 실행하세요.")
        return 0
    try:
        sid_hash = sanitize(sid)
    except ValueError:
        return 0

    config_path = _resolve_root() / "config.toml"
    try:
        ensure_config_file(config_path)
        config = load_config(config_path)
    except (OSError, ValueError):
        config = Config()
    lang = normalize_language(config.language)
    marker = Marker.load(sid_hash)

    arg = argv[0].strip() if argv else None

    # 무인자 — 상태 조회
    if arg is None:
        if marker.set_budget_remaining > 0:
            print(set_label(lang, "status_armed").format(
                n=marker.set_budget_remaining,
                total=marker.set_budget_total,
                time=_survive_time(config, marker.set_budget_remaining),
            ))
        else:
            print(set_label(lang, "status_none"))
        return 0

    if not arg.isdigit():                     # 음수/소수/문자 → usage
        print(set_label(lang, "usage"))
        return 0
    n = int(arg)

    if config.wake.arm == "always":
        print(set_label(lang, "always_noop"))
        return 0

    if n == 0:
        marker.set_budget_remaining = 0
        marker.set_budget_total = 0
        marker.save()
        print(set_label(lang, "cancelled"))
        return 0

    charged = min(n, config.max_refresh_count)
    marker.set_budget_remaining = charged
    marker.set_budget_total = charged
    marker.set_charged_at_ns = time.time_ns()
    marker.save()

    lines = [set_label(lang, "charged").format(
        n=charged, time=_survive_time(config, charged))]
    if charged < n:
        lines.append(set_label(lang, "capped_note").format(
            req=n, max=config.max_refresh_count))
    lines.append(set_label(lang, "session_only"))
    if marker.latest_fire == 0:
        lines.append(set_label(lang, "first_turn_note"))
    print("\n".join(lines))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

_STATUS_NAMES = {"cn:status", "cache-necromancer:cn:status"}
_SET_NAMES = {"cn:set", "cache-necromancer:cn:set"}


def _route(data: dict) -> tuple[Path, list[str]] | None:
    """매칭되는 (script, argv) 반환. 아니면 None."""
    cmd = (data.get("command_name") or "").strip()
    prompt = (data.get("prompt") or "").strip()
    if cmd:
        if cmd in _STATUS_NAMES:
            return _CN_STATUS, []
        if cmd in _SET_NAMES:
            args = prompt.split()[1:] if prompt else []
            return _CN_SET, args[:1]
        return None
    # command_name 비어있을 때만 prompt 문자열 fallback
    tokens = prompt.split()
    if not tokens:
        return None
    head = tokens[0]
    if head in {"/cn:status", "/cache-necromancer:cn:status"} and len(tokens) == 1:
        return _CN_STATUS, []
    if head in {"/cn:set", "/cache-necromancer:cn:set"}:
        return _CN_SET, tokens[1:2]
    return None
